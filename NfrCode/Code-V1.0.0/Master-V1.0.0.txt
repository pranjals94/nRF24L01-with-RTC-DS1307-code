// placing them too close can cause transmission failures
//Use capacitors closese to the power pins of NRF
// Dont use max power (test in low) from arduino 3.3 volt as it damages the 3.3 volt regulator
// the nfr module was failing when power wa supplied from modile charger, try good quality mobile chargers
// working flawlessly from laptop usb and battery power

#include <SPI.h>
#include <RF24.h>

RF24 radio(9, 10);

const byte address[6] = "NODE1";

const byte LED_PIN = A4;

void setup()
{
  Serial.begin(9600);
  delay(500);

  pinMode(LED_PIN, OUTPUT);
  digitalWrite(LED_PIN, LOW);

  if (!radio.begin())
  {
    Serial.println("NRF24 FAILED");

    while (1)
    {
      digitalWrite(LED_PIN, !digitalRead(LED_PIN));
      delay(200);
    }
  }

  radio.setChannel(100);
  radio.setDataRate(RF24_250KBPS);

  radio.setPALevel(RF24_PA_MAX);

  radio.setAutoAck(true);
  // radio.setRetries(15, 15);
  radio.setRetries(3, 5);//delay between retry's, no. of retrys

  radio.openWritingPipe(address);
  radio.stopListening();

  Serial.println("MASTER READY");
}

struct SensorData
{
    uint16_t temperature;
    uint16_t voltage;
    uint16_t current;
    uint16_t light;
};

SensorData data;

void loop()
{
    data.temperature++;
    data.voltage++;
    data.current++;
    data.light++;

  // Check local nRF24 before transmitting
  if (!radio.isChipConnected())
  {
    Serial.println("NRF24 NOT RESPONDING!");
    digitalWrite(LED_PIN, LOW);

    delay(100);
    return;
  }

    if (radio.write(&data, sizeof(data)))
    {
      digitalWrite(LED_PIN, HIGH);
      delay(50);
      digitalWrite(LED_PIN, LOW);
      Serial.println("Sensor data sent");
    }
    else
    {
        digitalWrite(LED_PIN, LOW);
        Serial.println("Transmission failed");
    }
    delay(500);
}
